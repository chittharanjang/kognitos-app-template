# DB Agent — Test Questions
#
# Source: app/guide (User Guide page) → lib/guide-queries.ts
# 18 categories, 100 queries. Keep this file in sync with the User Guide.
#
# Lines starting with `#` are comments. Blank lines separate groups.
# Each non-comment line is one question prefixed by its tag:
#   [TAG] question text
#
# Run individually:
#   npx tsx scripts/invoke-amaagent.ts "<question>"
#
# Batch-run (example):
#   while IFS= read -r line; do
#     [[ "$line" =~ ^#.*$ || -z "$line" ]] && continue
#     tag="${line%%]*}]"; q="${line#*] }"
#     echo "=== $tag ==="; npx tsx scripts/invoke-amaagent.ts "$q"
#   done < scripts/db-agent-test-questions.txt
#
# Tag legend:
#   YN     Yes/No (Single DB)              YNX    Yes/No (Cross-DB)
#   DSI    Direct Specific Information     COL    Specific Column Requests
#   LIST   Full Table / List / File        FIDO   Single DB - FIDO
#   B3     Single DB - WealthX             PS     Single DB - Profile Status
#   J2     Cross-DB Joins (Two DBs)        B4     Cross-DB Joins (Three DBs)
#   IRA    IRA Account Type Handling       MULTI  Multi-Question Emails
#   AMB    Ambiguous / Poorly Structured   PARA   Long Paragraph Queries
#   NDF    No Data Found Scenarios         GOV    Data Governance / Security
#   Q      Composite / Complex Business    N      Aggregation / Count
